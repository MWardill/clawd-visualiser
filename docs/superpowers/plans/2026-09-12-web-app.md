# Web App (`apps/web`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `apps/web` show the fleet on the panes scene, fed by the simulated source in the browser or by the local agent's SSE stream when one answers, with status bars, a waiting count in the tab title, and a detail sheet whose only content is the redacted ticker in xterm.js plus a focus button.

**Architecture:** A `Feed` interface with two implementations (live over `EventSource`, simulated over `@clawd/core`'s reducers) and a pure connection policy that picks between them. One client component owns the feed and the selection; the canvas component mounts `mountPanesScene` from `@clawd/scene` and forwards snapshots; the sheet loads xterm.js on demand. Nothing but `Snapshot`, `TickerEvent` and `Health` is ever received, and every live message is validated against the strict wire schemas.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind v4, shadcn/ui (base-nova style on Base UI; `sheet`, `button`, `badge` already generated in `src/components/ui/`), `@xterm/xterm` 6 with `@xterm/addon-fit`, Vitest 5, TypeScript 5 strict, pnpm 12, Node 24.

```ts
/** Local wall-clock `HH:MM:SS` for a timestamp; dashes when it does not parse. */
export function hhmmss(when: string | number | Date): string {
  const d = new Date(when);
  if (Number.isNaN(d.getTime())) return "--:--:--";
  return [d.getHours(), d.getMinutes(), d.getSeconds()].map((n) => String(n).padStart(2, "0")).join(":");
}
```

```ts
import type { Snapshot } from "@clawd/core";
import type { FeedStatus } from "./feed/types";

export const APP_TITLE = "Clawd Visualiser";

export type Counts = { sessions: number; waiting: number; subagents: number; panes: number };

/** Live sessions only: a done session lingers a few seconds for its exit and is not counted. */
export function countSnapshot(s: Snapshot | null): Counts {
  const counts: Counts = { sessions: 0, waiting: 0, subagents: 0, panes: s?.panes.length ?? 0 };
  for (const session of s?.sessions ?? []) {
    if (session.state === "done") continue;
    counts.sessions++;
    if (session.state === "waiting") counts.waiting++;
    counts.subagents += session.subagents;
  }
  return counts;
}

/** The tab title carries the waiting count while there is one (FR-24). */
export function tabTitle(waiting: number): string {
  return waiting > 0 ? `(${waiting}) ${APP_TITLE}` : APP_TITLE;
}

/** The canvas's accessible name. */
export function sceneLabel(c: Counts): string {
  return `${c.sessions} sessions in ${c.panes} panes, ${c.waiting} waiting`;
}

const REASONS = { "no-agent": "no agent", "lost-agent": "agent lost", forced: "forced" } as const;

/** The source badge in the top bar. */
export function sourceBadge(status: FeedStatus): string {
  switch (status.mode) {
    case "probing":
      return "probing for agent";
    case "live": {
      if (!status.connected) return "live · reconnecting";
      const host = hostOf(status.agentUrl);
      return status.rejected > 0 ? `live · ${host} · ${status.rejected} rejected` : `live · ${host}`;
    }
    case "simulated":
      return `simulated · ${REASONS[status.reason]}`;
    default:
      status satisfies never;
      return "";
  }
}

function hostOf(url: string): string {
  try {
    return new URL(url).host;
  } catch {
    return url;
  }
}
```

```ts
// One ticker line per event, coloured for xterm.js. The only strings are the fields TickerEvent
// carries: a kind, an activity, a tool name, a subagent type, a state, a reason. Nothing else exists to show.
import type { Activity, SessionPhase, TickerEvent } from "@clawd/core";
import { hhmmss } from "../time";

const ESC = "\u001b[";
const RESET = `${ESC}0m`;
const paint = (code: string, s: string): string => `${ESC}${code}m${s}${RESET}`;
const dim = (s: string): string => paint("2", s);

/** ANSI codes by activity; the terminal theme binds red to Clawd orange for spawn. */
const ACTIVITY_CODES: Record<Activity, string> = { read: "34", edit: "32", run: "33", web: "35", spawn: "31", think: "2", other: "2" };
const STATE_CODES: Record<SessionPhase, string> = { working: "32", waiting: "33", idle: "2", asleep: "34", done: "35" };

const KIND_WIDTH = 14;

export function formatTicker(e: TickerEvent): string {
  const time = dim(hhmmss(e.t));
  const kind = e.kind.padEnd(KIND_WIDTH);
  switch (e.kind) {
    case "tool.start":
    case "tool.end":
    case "permission": {
      const parts = [time, e.kind === "permission" ? paint("33", kind) : kind];
      if (e.activity) parts.push(paint(ACTIVITY_CODES[e.activity], e.activity.padEnd(5)));
      if (e.tool) parts.push(e.tool.padEnd(12));
      if (e.durationMs !== undefined) parts.push(dim(`${Math.round(e.durationMs)}ms`));
      return parts.join(" ").trimEnd();
    }
    case "state":
      return `${time} ${kind} ${e.state ? paint(STATE_CODES[e.state], e.state) : ""}`.trimEnd();
    case "subagent.start":
    case "subagent.stop":
      return `${time} ${kind} ${e.agentType ?? ""}`.trimEnd();
    case "context":
      return `${time} ${kind} ${e.contextPct !== undefined ? `${e.contextPct}%` : ""}`.trimEnd();
    case "session.end":
      return `${time} ${kind} ${e.reason ?? ""}`.trimEnd();
    case "session.start":
    case "prompt":
    case "stop":
      return `${time} ${kind}`.trimEnd();
    default:
      e.kind satisfies never;
      return `${time} ${String(e.kind)}`;
  }
}

/** Removes the colour sequences; for tests and for plain-text fallbacks. */
export function stripAnsi(s: string): string {
  return s.replace(/\u001b\[[0-9;]*m/g, "");
}
```

Note the colour sequences wrap padded fields, so `trimEnd()` on a line ending in a reset code leaves trailing spaces inside the sequence; that is harmless in a terminal and the tests split on whitespace.

```ts
// The three HTTP calls the app makes to the agent, each validated with the shared schemas. `fetch`
// is injected so tests need no network.
import { CommandSchema, DEFAULT_AGENT_URL, ENDPOINTS, HealthSchema, TickerEventSchema, type Command, type TickerEvent } from "@clawd/core";

export type FetchLike = (input: string, init?: RequestInit) => Promise<Response>;

const defaultFetch: FetchLike = (input, init) => fetch(input, init);

/** The agent's base URL: NEXT_PUBLIC_AGENT_URL at build time, else the contract's default. No trailing slash. */
export function agentUrl(): string {
  return (process.env.NEXT_PUBLIC_AGENT_URL || DEFAULT_AGENT_URL).replace(/\/+$/, "");
}

export const PROBE_TIMEOUT_MS = 1500;

/** True when an agent answers the health endpoint with a valid Health body within the timeout. */
export async function probeHealth(base: string, fetchFn: FetchLike = defaultFetch, timeoutMs = PROBE_TIMEOUT_MS): Promise<boolean> {
  try {
    const res = await fetchFn(`${base}${ENDPOINTS.health}`, { signal: AbortSignal.timeout(timeoutMs) });
    if (!res.ok) return false;
    return HealthSchema.safeParse(await res.json()).success;
  } catch {
    return false;
  }
}

/** The last ticker lines the agent kept for a session; anything that does not validate is dropped. */
export async function fetchBackfill(base: string, alias: string, fetchFn: FetchLike = defaultFetch): Promise<TickerEvent[]> {
  try {
    const res = await fetchFn(`${base}${ENDPOINTS.ticker(encodeURIComponent(alias))}`);
    if (!res.ok) return [];
    const body: unknown = await res.json();
    if (!Array.isArray(body)) return [];
    const out: TickerEvent[] = [];
    for (const item of body) {
      const parsed = TickerEventSchema.safeParse(item);
      if (parsed.success) out.push(parsed.data);
    }
    return out;
  } catch {
    return [];
  }
}

/** Sends a Command; true when the agent accepted it with 202. */
export async function sendCommand(base: string, command: Command, fetchFn: FetchLike = defaultFetch): Promise<boolean> {
  const parsed = CommandSchema.safeParse(command);
  if (!parsed.success) return false;
  try {
    const res = await fetchFn(`${base}${ENDPOINTS.command}`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(parsed.data),
    });
    return res.status === 202;
  } catch {
    return false;
  }
}
```

```ts
// Which feed to run, as a pure reducer with effects, so the fallback rules are testable without
// timers. The hook runs the effects: probes, feed starts, and the two timers.
//
//   probing --probe ok--> live --stream down for GRACE_MS--> simulated (lost-agent)
//      +--probe fails--> simulated (no-agent) --re-probe every REPROBE_MS, ok--> live
import type { FeedStatus, SimulatedReason } from "./types";

export type SourceMode = "auto" | "simulated" | "live";
export type PolicyPhase = "probing" | "live" | "simulated";

export type PolicyState = {
  mode: SourceMode;
  phase: PolicyPhase;
  connected: boolean;
  rejected: number;
  reason: SimulatedReason | null;
};

export type PolicyEvent =
  | { type: "start" }
  | { type: "probe"; ok: boolean }
  | { type: "connection"; connected: boolean; rejected: number }
  | { type: "grace-elapsed" }
  | { type: "reprobe-due" };

export type Effect = "probe" | "start-live" | "start-simulated" | "schedule-reprobe" | "schedule-grace" | "cancel-grace";

/** How often simulated mode looks for an agent. */
export const REPROBE_MS = 10_000;
/** How long a live stream may stay down before the app falls back. */
export const GRACE_MS = 10_000;

/** `?source=simulated` forces the simulation with no probing; `?source=live` disables the fallback. */
export function parseSourceMode(search: string): SourceMode {
  const value = new URLSearchParams(search).get("source");
  return value === "simulated" || value === "live" ? value : "auto";
}

export function initialPolicy(mode: SourceMode): PolicyState {
  return { mode, phase: "probing", connected: false, rejected: 0, reason: null };
}

const none: Effect[] = [];

export function reducePolicy(s: PolicyState, e: PolicyEvent): { state: PolicyState; effects: Effect[] } {
  switch (e.type) {
    case "start":
      if (s.mode === "simulated") return { state: { ...s, phase: "simulated", reason: "forced" }, effects: ["start-simulated"] };
      if (s.mode === "live") return { state: { ...s, phase: "live", connected: false }, effects: ["start-live"] };
      return { state: { ...s, phase: "probing" }, effects: ["probe"] };
    case "probe":
      if (s.mode !== "auto" || s.phase === "live") return { state: s, effects: none };
      if (e.ok) return { state: { ...s, phase: "live", connected: false, rejected: 0, reason: null }, effects: ["start-live", "schedule-grace"] };
      if (s.phase === "probing") return { state: { ...s, phase: "simulated", reason: "no-agent" }, effects: ["start-simulated", "schedule-reprobe"] };
      return { state: s, effects: ["schedule-reprobe"] };
    case "reprobe-due":
      return s.mode === "auto" && s.phase === "simulated" ? { state: s, effects: ["probe"] } : { state: s, effects: none };
    case "connection":
      if (s.phase !== "live") return { state: s, effects: none };
      if (e.connected) return { state: { ...s, connected: true, rejected: e.rejected }, effects: ["cancel-grace"] };
      return { state: { ...s, connected: false, rejected: e.rejected }, effects: s.mode === "auto" ? ["schedule-grace"] : none };
    case "grace-elapsed":
      if (s.mode !== "auto" || s.phase !== "live" || s.connected) return { state: s, effects: none };
      return { state: { ...s, phase: "simulated", reason: "lost-agent", connected: false }, effects: ["start-simulated", "schedule-reprobe"] };
    default:
      e satisfies never;
      return { state: s, effects: none };
  }
}

export function toFeedStatus(s: PolicyState, agentUrl: string): FeedStatus {
  switch (s.phase) {
    case "probing":
      return { mode: "probing" };
    case "live":
      return { mode: "live", agentUrl, connected: s.connected, rejected: s.rejected };
    case "simulated":
      return { mode: "simulated", reason: s.reason ?? "no-agent" };
    default:
      s.phase satisfies never;
      return { mode: "probing" };
  }
}
```

```ts
// The browser's stand-in for the agent: the core simulated source through the same reducers the
// agent runs, a snapshot at most every SNAPSHOT_MIN_INTERVAL_MS, and a ring of ticker lines per
// alias for the detail pane's backfill. Session ids and pane keys never leave this module.
import {
  SIMULATED_CONFIG,
  SNAPSHOT_MIN_INTERVAL_MS,
  TICKER_BACKFILL,
  applyEvent,
  createSimulatedSource,
  deriveSnapshot,
  emptyAliasMap,
  emptyFleet,
  ensureSessionAlias,
  tickFleet,
  type AliasMap,
  type Fleet,
  type RedactedEvent,
  type SimulatedSourceOptions,
  type TickerEvent,
} from "@clawd/core";
import type { Feed, FeedListener } from "./types";

export type SimulatedFeedOptions = {
  seed?: number;
  sessions?: number;
  now?: () => number;
  setInterval?: (fn: () => void, ms: number) => unknown;
  clearInterval?: (handle: unknown) => void;
  /** How often the fleet's timers are applied and a snapshot derived. Default SNAPSHOT_MIN_INTERVAL_MS. */
  tickMs?: number;
};

export function createSimulatedFeed(opts: SimulatedFeedOptions = {}): Feed {
  const now = opts.now ?? (() => Date.now());
  const setI = opts.setInterval ?? ((fn, ms) => globalThis.setInterval(fn, ms));
  const clearI = opts.clearInterval ?? ((h) => globalThis.clearInterval(h as ReturnType<typeof globalThis.setInterval>));
  const rings = new Map<string, TickerEvent[]>();

  return {
    start(listener: FeedListener) {
      let fleet: Fleet = emptyFleet();
      let aliases: AliasMap = emptyAliasMap();
      let lastKey = "";

      const publish = (t: number): void => {
        const r = deriveSnapshot(fleet, aliases, t);
        aliases = r.aliases;
        const key = JSON.stringify([r.snapshot.panes, r.snapshot.sessions]);
        if (key === lastKey) return;
        lastKey = key;
        listener.onSnapshot(r.snapshot);
      };

      const sink = (e: RedactedEvent): void => {
        const r = applyEvent(fleet, e, e.t, SIMULATED_CONFIG);
        fleet = r.fleet;
        if (r.ticker.length === 0) return;
        const ar = ensureSessionAlias(aliases, e.sessionId);
        aliases = ar.map;
        const ring = rings.get(ar.alias) ?? [];
        rings.set(ar.alias, ring);
        for (const line of r.ticker) {
          ring.push(line);
          if (ring.length > TICKER_BACKFILL) ring.shift();
          listener.onTicker(ar.alias, line);
        }
      };

      const sourceOpts: SimulatedSourceOptions = { now, setInterval: setI, clearInterval: clearI };
      if (opts.seed !== undefined) sourceOpts.seed = opts.seed;
      if (opts.sessions !== undefined) sourceOpts.sessions = opts.sessions;
      const stopSource = createSimulatedSource(sourceOpts).start(sink);
      publish(now());
      const handle = setI(() => {
        const t = now();
        fleet = tickFleet(fleet, t, SIMULATED_CONFIG);
        publish(t);
      }, opts.tickMs ?? SNAPSHOT_MIN_INTERVAL_MS);
      listener.onConnection?.(true, 0);

      return () => {
        stopSource();
        clearI(handle);
      };
    },
    backfill(alias) {
      return Promise.resolve([...(rings.get(alias) ?? [])]);
    },
    focus() {
      return Promise.resolve(false);
    },
  };
}
```

```ts
// The agent's SSE stream. Every message is validated against the strict wire schemas and dropped
// otherwise, so nothing outside the contract can reach a component. EventSource reconnects by
// itself; the policy decides when a long outage means falling back.
import { ENDPOINTS, STREAM_EVENTS, SnapshotSchema, StreamTickerMessageSchema } from "@clawd/core";
import { fetchBackfill, sendCommand, type FetchLike } from "./agent";
import type { Feed } from "./types";

export type EventSourceLike = {
  addEventListener(type: string, listener: (e: MessageEvent) => void): void;
  close(): void;
};

export type LiveFeedDeps = {
  EventSource?: new (url: string) => EventSourceLike;
  fetch?: FetchLike;
};

function parseJson(text: unknown): unknown {
  if (typeof text !== "string") return null;
  try {
    return JSON.parse(text);
  } catch {
    return null;
  }
}

export function createLiveFeed(base: string, deps: LiveFeedDeps = {}): Feed {
  const Source = deps.EventSource ?? (EventSource as unknown as new (url: string) => EventSourceLike);
  const fetchFn: FetchLike = deps.fetch ?? ((input, init) => fetch(input, init));

  return {
    start(listener) {
      let rejected = 0;
      const source = new Source(`${base}${ENDPOINTS.stream}`);
      const report = (connected: boolean): void => listener.onConnection?.(connected, rejected);
      source.addEventListener("open", () => report(true));
      source.addEventListener("error", () => report(false));
      source.addEventListener(STREAM_EVENTS.snapshot, (e) => {
        const parsed = SnapshotSchema.safeParse(parseJson(e.data));
        if (parsed.success) {
          listener.onSnapshot(parsed.data);
        } else {
          rejected++;
          report(true);
        }
      });
      source.addEventListener(STREAM_EVENTS.ticker, (e) => {
        const parsed = StreamTickerMessageSchema.safeParse(parseJson(e.data));
        if (parsed.success) {
          listener.onTicker(parsed.data.alias, parsed.data.event);
        } else {
          rejected++;
          report(true);
        }
      });
      return () => source.close();
    },
    backfill(alias) {
      return fetchBackfill(base, alias, fetchFn);
    },
    focus(alias) {
      return sendCommand(base, { type: "focus", alias }, fetchFn);
    },
  };
}
```

```ts
"use client";

// Runs the connection policy and whichever feed it picks. Components get the latest snapshot,
// a status for the badge, a generation counter that changes when the feed changes (so the scene
// can clear), and the ticker subscription the detail sheet needs.
import type { Snapshot, TickerEvent } from "@clawd/core";
import { useCallback, useEffect, useMemo, useRef, useState, useSyncExternalStore } from "react";
import { agentUrl, probeHealth } from "@/lib/feed/agent";
import { createLiveFeed } from "@/lib/feed/live";
import { GRACE_MS, REPROBE_MS, initialPolicy, parseSourceMode, reducePolicy, toFeedStatus, type Effect, type PolicyEvent, type PolicyState } from "@/lib/feed/policy";
import { createSimulatedFeed } from "@/lib/feed/simulated";
import type { Feed, FeedStatus } from "@/lib/feed/types";

export type TickerSubscriber = (e: TickerEvent) => void;

export type FeedApi = {
  snapshot: Snapshot | null;
  status: FeedStatus;
  /** Increments each time a different feed starts; the scene clears on it. */
  generation: number;
  canFocus: boolean;
  subscribeTicker: (alias: string, fn: TickerSubscriber) => () => void;
  backfill: (alias: string) => Promise<TickerEvent[]>;
  focus: (alias: string) => Promise<boolean>;
};

const subscribeToNothing = (): (() => void) => () => {};
const readSearch = (): string => window.location.search;
const readServerSearch = (): string => "";

export function useFeed(): FeedApi {
  const search = useSyncExternalStore(subscribeToNothing, readSearch, readServerSearch);
  const mode = parseSourceMode(search);
  const base = useMemo(agentUrl, []);
  const [snapshot, setSnapshot] = useState<Snapshot | null>(null);
  const [status, setStatus] = useState<FeedStatus>({ mode: "probing" });
  const [generation, setGeneration] = useState(0);
  const feedRef = useRef<Feed | null>(null);
  const subscribersRef = useRef<Map<string, Set<TickerSubscriber>>>(new Map());

  useEffect(() => {
    let state: PolicyState = initialPolicy(mode);
    let stopFeed: (() => void) | null = null;
    let grace: ReturnType<typeof setTimeout> | null = null;
    let reprobe: ReturnType<typeof setTimeout> | null = null;
    let disposed = false;

    const startFeed = (feed: Feed): void => {
      stopFeed?.();
      feedRef.current = feed;
      setSnapshot(null);
      setGeneration((g) => g + 1);
      stopFeed = feed.start({
        onSnapshot: (s) => setSnapshot(s),
        onTicker: (alias, e) => {
          for (const fn of subscribersRef.current.get(alias) ?? []) fn(e);
        },
        onConnection: (connected, rejected) => dispatch({ type: "connection", connected, rejected }),
      });
    };

    const run = (effect: Effect): void => {
      switch (effect) {
        case "probe":
          void probeHealth(base).then((ok) => dispatch({ type: "probe", ok }));
          break;
        case "start-live":
          startFeed(createLiveFeed(base));
          break;
        case "start-simulated":
          startFeed(createSimulatedFeed());
          break;
        case "schedule-reprobe":
          if (reprobe) clearTimeout(reprobe);
          reprobe = setTimeout(() => dispatch({ type: "reprobe-due" }), REPROBE_MS);
          break;
        case "schedule-grace":
          if (grace) clearTimeout(grace);
          grace = setTimeout(() => dispatch({ type: "grace-elapsed" }), GRACE_MS);
          break;
        case "cancel-grace":
          if (grace) clearTimeout(grace);
          grace = null;
          break;
        default:
          effect satisfies never;
      }
    };

    function dispatch(e: PolicyEvent): void {
      if (disposed) return;
      const r = reducePolicy(state, e);
      state = r.state;
      setStatus(toFeedStatus(state, base));
      for (const effect of r.effects) run(effect);
    }

    // Deferred so no state setter runs synchronously inside the effect body.
    queueMicrotask(() => dispatch({ type: "start" }));

    return () => {
      disposed = true;
      stopFeed?.();
      feedRef.current = null;
      if (grace) clearTimeout(grace);
      if (reprobe) clearTimeout(reprobe);
    };
  }, [mode, base]);

  const subscribeTicker = useCallback((alias: string, fn: TickerSubscriber) => {
    const map = subscribersRef.current;
    const set = map.get(alias) ?? new Set<TickerSubscriber>();
    map.set(alias, set);
    set.add(fn);
    return () => {
      set.delete(fn);
      if (set.size === 0) map.delete(alias);
    };
  }, []);
  const backfill = useCallback((alias: string) => feedRef.current?.backfill(alias) ?? Promise.resolve([]), []);
  const focus = useCallback((alias: string) => feedRef.current?.focus(alias) ?? Promise.resolve(false), []);

  return { snapshot, status, generation, canFocus: status.mode === "live" && status.connected, subscribeTicker, backfill, focus };
}
```

```tsx
  const snapshotRef = useRef<Snapshot | null>(null);

    sceneRef.current?.clear();
  }, [generation]);

    const canvas = canvasRef.current;
    const host = canvas?.parentElement;
    if (!canvas || !host) return;

      observer.disconnect();
      scene?.destroy();
      sceneRef.current = null;
```

```tsx
  backfill: (alias: string) => Promise<TickerEvent[]>;

export default function TickerTerminal({ alias, subscribe, backfill }: Props) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const term = new Terminal({
      theme: THEME,
      fontFamily: getComputedStyle(el).fontFamily,
      fontSize: 12,
      lineHeight: 1.2,
      disableStdin: true,
      cursorBlink: false,
      cursorStyle: "underline",
      cursorInactiveStyle: "none",
      convertEol: true,
      scrollback: 2000,
    });
    const fit = new FitAddon();
    term.loadAddon(fit);
    term.open(el);
    fit.fit();

    let ready = false;
    let cancelled = false;
    const queue: TickerEvent[] = [];
    const write = (e: TickerEvent): void => term.writeln(formatTicker(e));
    const unsubscribe = subscribe(alias, (e) => {
      if (ready) write(e);
      else queue.push(e);
    });
    void backfill(alias).then((lines) => {
      if (cancelled) return;
      for (const line of lines) write(line);
      for (const line of queue) write(line);
      queue.length = 0;
      ready = true;
    });
    const observer = new ResizeObserver(() => fit.fit());
    observer.observe(el);

    return () => {
      cancelled = true;
      unsubscribe();
      observer.disconnect();
      term.dispose();
    };
  }, [alias, subscribe, backfill]);

  return <div ref={ref} className="h-full w-full" />;
}
```

```tsx
                  {session.subagents > 0 ? <span>×{session.subagents} subagents</span> : null}
                  {session.contextPct !== null ? <span>ctx {session.contextPct}%</span> : null}
                </>
              ) : (
                <span>session ended</span>
              )}
            </SheetDescription>
          </div>
        </div>
        <div className="mt-2 flex items-center gap-3">
          <Button size="sm" variant="outline" disabled={!canFocus || !session || result === "sending"} onClick={onFocus}>
            focus window
          </Button>
          <span className="text-xs text-muted-foreground">{hint}</span>
        </div>
      </SheetHeader>
      <div className="min-h-0 flex-1 bg-background p-2">
        <TickerTerminal alias={alias} subscribe={subscribeTicker} backfill={backfill} />
      </div>
    </>
  );
}
```
