# Core Contract (`packages/core`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `@clawd/core`, the shared contract between the local agent and the web app: wire schemas, the redacted event model, ingest redaction, the state machine, aliases, the source interface with simulated and replay sources, and recorded fixtures, all under test.

**Architecture:** Pure TypeScript with `zod` as the only runtime dependency. Raw Claude Code payloads enter through one function, `redact()`, which produces `RedactedEvent`s; pure reducers turn events into a `Fleet`; `deriveSnapshot()` is the only place a wire `Snapshot` is made and the only place local identifiers are swapped for aliases. Synthetic sources emit the same `RedactedEvent`s, so simulated mode exercises the same reducers as live mode.

**Tech Stack:** TypeScript 5 strict (`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`), zod 4, Vitest 5, pnpm 12 workspace, Node 24. `tsx` (dev only) runs the one-off fixture script.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-09-12-core-contract-design.md`. Read it before starting.
- `packages/core/src` contains no I/O and no `node:` imports: no `fs`, `http`, or DOM. `sources/` may default to the global timers but must accept injected ones. Test files may import `node:fs` and `node:path`.
- `zod` is the only runtime dependency of `@clawd/core`. Dev dependencies may be added (`@types/node`, `tsx`).
- Every reducer is pure: state in, state out, `now` (epoch ms) passed in. Never call `Date.now()` inside `src/` except as an injectable default in `sources/`.
- Wire schemas use `.strict()`; `ext` is the only extension field.
- The forbidden list in `ingest/redact.ts` must contain exactly: `prompt`, `tool_input`, `tool_response`, `last_assistant_message`, `transcript_path`, `agent_transcript_path`, `session_name`, `cwd`, `workspace`.
- Commit locally after each task with a one-line message and no trailers. **Never push**; the user pushes.
- Run commands from the repo root `C:\mw\visualiser` unless stated. Package filter: `pnpm --filter @clawd/core <script>`.
- Line endings LF (enforced by `.gitattributes`).

```ts
export const SessionPhaseSchema = z.enum(["working", "waiting", "idle", "asleep", "done"]);
export type SessionPhase = z.infer<typeof SessionPhaseSchema>;

/** What a working session is doing. `think` is the gap between tool calls. */

export type Activity = z.infer<typeof ActivitySchema>;

const Ext = z.record(z.string(), z.unknown()).optional();

export const PaneSnapshotSchema = z
  .object({
    pane: z.string().min(1),
    label: z.string().min(1),
    collision: z.boolean(),
  })
  .strict();
export type PaneSnapshot = z.infer<typeof PaneSnapshotSchema>;

export const SessionSnapshotSchema = z
  .object({
    alias: z.string().min(1),
    label: z.string().min(1),
    pane: z.string().min(1),
    state: SessionPhaseSchema,
    activity: ActivitySchema.nullable(),
    subagents: z.number().int().nonnegative(),
    contextPct: z.number().min(0).max(100).nullable(),
    since: z.iso.datetime(),
    updatedAt: z.iso.datetime(),
    ext: Ext,
  })
  .strict();
export type SessionSnapshot = z.infer<typeof SessionSnapshotSchema>;

export const SnapshotSchema = z
  .object({
    version: z.literal(1),
    generatedAt: z.iso.datetime(),
    panes: z.array(PaneSnapshotSchema),
    sessions: z.array(SessionSnapshotSchema),
    ext: Ext,
  })
  .strict();
export type Snapshot = z.infer<typeof SnapshotSchema>;
```

```ts
import { z } from "zod";

export const EVENT_SCHEMA_VERSION = 1 as const;

/** Activity categories a tool call maps to. `think` is not here: it is derived between calls. */
export const ToolCategorySchema = z.enum(["read", "edit", "run", "web", "spawn", "other"]);
export type ToolCategory = z.infer<typeof ToolCategorySchema>;

const base = {
  v: z.literal(EVENT_SCHEMA_VERSION),
  t: z.number().int().nonnegative(),
  sessionId: z.string().min(1),
  paneKey: z.string().min(1),
  agentId: z.string().min(1).optional(),
};

const tool = { tool: z.string().min(1), category: ToolCategorySchema };

/**
 * The internal event type: what remains of a Claude Code payload after ingest. Carries the
 * session id and a hashed working directory so reducers can tell sessions apart; those two
 * never reach a snapshot.
 */
export const RedactedEventSchema = z.discriminatedUnion("kind", [
  z.object({ ...base, kind: z.literal("session.start"), source: z.string().optional() }).strict(),
  z.object({ ...base, kind: z.literal("session.end"), reason: z.string() }).strict(),
  z.object({ ...base, kind: z.literal("prompt") }).strict(),
  z.object({ ...base, kind: z.literal("tool.start"), ...tool, toolUseId: z.string().min(1) }).strict(),
  z.object({ ...base, kind: z.literal("tool.end"), ...tool, toolUseId: z.string().min(1), durationMs: z.number().nonnegative() }).strict(),
  z.object({ ...base, kind: z.literal("permission.request"), ...tool, toolUseId: z.string().min(1).optional() }).strict(),
  z.object({ ...base, kind: z.literal("permission.denied"), tool: z.string().min(1).optional() }).strict(),
  z.object({ ...base, kind: z.literal("stop"), backgroundTasks: z.number().int().nonnegative() }).strict(),
  z.object({ ...base, agentId: z.string().min(1), kind: z.literal("subagent.start"), agentType: z.string().min(1) }).strict(),
  z.object({ ...base, agentId: z.string().min(1), kind: z.literal("subagent.stop"), agentType: z.string().min(1) }).strict(),
  z
    .object({
      ...base,
      kind: z.literal("status"),
      contextPct: z.number().min(0).max(100).optional(),
      costUsd: z.number().nonnegative().optional(),
      rateLimit5h: z.number().min(0).max(100).optional(),
      rateLimit7d: z.number().min(0).max(100).optional(),
      model: z.string().optional(),
    })
    .strict(),
  z.object({ ...base, kind: z.literal("presence"), alive: z.boolean() }).strict(),
]);
export type RedactedEvent = z.infer<typeof RedactedEventSchema>;
export type RedactedEventKind = RedactedEvent["kind"];
```

```ts
/** Lower-case, forward slashes, no trailing slash (a bare "/" stays "/"). */
export function normalisePath(p: string): string {
  const s = p.replace(/\\/g, "/").toLowerCase();
  return s.length > 1 ? s.replace(/\/+$/, "") : s;
}

const FNV_OFFSET = 0xcbf29ce484222325n;
const FNV_PRIME = 0x100000001b3n;
const MASK64 = (1n << 64n) - 1n;

/**
 * 64-bit FNV-1a of the normalised path, as 16 hex characters. An internal key so the path never
 * appears in a log line; the snapshot carries a pane alias, never this.
 *
 * Hashes UTF-16 code units rather than UTF-8 bytes so it needs no TextEncoder (the package has no
 * DOM or Node types). For ASCII paths this is exactly FNV-1a over the bytes.
 */
export function paneKey(cwd: string): string {
  const s = normalisePath(cwd);
  let h = FNV_OFFSET;
  for (let i = 0; i < s.length; i++) {
    h ^= BigInt(s.charCodeAt(i));
    h = (h * FNV_PRIME) & MASK64;
  }
  return h.toString(16).padStart(16, "0");
}
```

```ts
import { paneKey } from "../alias/pane";
import { RedactedEventSchema, type RedactedEvent } from "../schema/events";
import { RawHookSchema, RawStatusLineSchema, type RawHook, type RawStatusLine } from "../schema/raw";
import { isAllowed } from "./allowlist";
import { categoryFor, normaliseTool } from "./categories";

export type IngestContext = { now: number; allowlist: readonly string[] };

/** Payload fields that must never survive ingest. Tested against every fixture. */
export const FORBIDDEN_KEYS = [
  "prompt",
  "tool_input",
  "tool_response",
  "last_assistant_message",
  "transcript_path",
  "agent_transcript_path",
  "session_name",
  "cwd",
  "workspace",
] as const;

/**
 * Turn a raw hook or status line payload into a redacted event, or null when the payload is not
 * ours to keep: outside the allowlist, an internal helper subagent, an unmodelled event, or junk.
 */
export function redact(raw: unknown, ctx: IngestContext): RedactedEvent | null {
  if (typeof raw !== "object" || raw === null) return null;
  const hook = RawHookSchema.safeParse(raw);
  if (hook.success) return redactHook(hook.data, ctx);
  const status = RawStatusLineSchema.safeParse(raw);
  if (status.success) return redactStatus(status.data, ctx);
  return null;
}

function validate(event: unknown): RedactedEvent | null {
  const parsed = RedactedEventSchema.safeParse(event);
  return parsed.success ? parsed.data : null;
}

function redactHook(p: RawHook, ctx: IngestContext): RedactedEvent | null {
  if (!isAllowed(p.cwd, ctx.allowlist)) return null;
  const base = { v: 1 as const, t: ctx.now, sessionId: p.session_id, paneKey: paneKey(p.cwd), ...(p.agent_id ? { agentId: p.agent_id } : {}) };
  const rawTool = p.tool_name ?? "unknown";
  const toolFields = { tool: normaliseTool(rawTool), category: categoryFor(rawTool) };
  switch (p.hook_event_name) {
    case "SessionStart":
      return validate({ ...base, kind: "session.start", ...(p.source ? { source: p.source } : {}) });
    case "SessionEnd":
      return validate({ ...base, kind: "session.end", reason: p.reason ?? "unknown" });
    case "UserPromptSubmit":
      return validate({ ...base, kind: "prompt" });
    case "PreToolUse":
      if (!p.tool_use_id) return null;
      return validate({ ...base, kind: "tool.start", ...toolFields, toolUseId: p.tool_use_id });
    case "PostToolUse":
      if (!p.tool_use_id) return null;
      return validate({ ...base, kind: "tool.end", ...toolFields, toolUseId: p.tool_use_id, durationMs: Math.max(0, p.duration_ms ?? 0) });
    case "PermissionRequest":
      return validate({ ...base, kind: "permission.request", ...toolFields, ...(p.tool_use_id ? { toolUseId: p.tool_use_id } : {}) });
    case "PermissionDenied":
      return validate({ ...base, kind: "permission.denied", ...(p.tool_name ? { tool: normaliseTool(p.tool_name) } : {}) });
    case "Stop":
      return validate({ ...base, kind: "stop", backgroundTasks: p.background_tasks?.length ?? 0 });
    case "SubagentStart":
    case "SubagentStop":
      // Internal helpers (the session-naming helper) have an empty agent_type. Drop them.
      if (!p.agent_id || !p.agent_type) return null;
      return validate({ ...base, agentId: p.agent_id, kind: p.hook_event_name === "SubagentStart" ? "subagent.start" : "subagent.stop", agentType: p.agent_type });
    default:
      return null;
  }
}

function redactStatus(p: RawStatusLine, ctx: IngestContext): RedactedEvent | null {
  if (!isAllowed(p.cwd, ctx.allowlist)) return null;
  const pct = (v: number | undefined): number | undefined => (typeof v === "number" ? Math.min(100, Math.max(0, v)) : undefined);
  const contextPct = pct(p.context_window.used_percentage);
  const costUsd = p.cost?.total_cost_usd;
  const rateLimit5h = pct(p.rate_limits?.five_hour?.used_percentage);
  const rateLimit7d = pct(p.rate_limits?.seven_day?.used_percentage);
  const model = p.model?.display_name;
  return validate({
    v: 1 as const,
    t: ctx.now,
    sessionId: p.session_id,
    paneKey: paneKey(p.cwd),
    kind: "status",
    ...(contextPct !== undefined ? { contextPct } : {}),
    ...(typeof costUsd === "number" ? { costUsd: Math.max(0, costUsd) } : {}),
    ...(rateLimit5h !== undefined ? { rateLimit5h } : {}),
    ...(rateLimit7d !== undefined ? { rateLimit7d } : {}),
    ...(model ? { model } : {}),
  });
}

/**
 * Walk any value; return the first forbidden key name found, or the first extra string found
 * inside any string value, or null. Used by the redaction tests and by the agent's self-check.
 */
export function containsForbidden(value: unknown, extraStrings: readonly string[] = []): string | null {
  if (typeof value === "string") {
    for (const s of extraStrings) if (s && value.includes(s)) return s;
    return null;
  }
  if (Array.isArray(value)) {
    for (const v of value) {
      const hit = containsForbidden(v, extraStrings);
      if (hit) return hit;
    }
    return null;
  }
  if (typeof value === "object" && value !== null) {
    for (const [k, v] of Object.entries(value)) {
      if ((FORBIDDEN_KEYS as readonly string[]).includes(k)) return k;
      const hit = containsForbidden(v, extraStrings);
      if (hit) return hit;
    }
  }
  return null;
}
```

```ts
import type { RedactedEvent } from "../schema/events";
import type { Activity, SessionPhase } from "../schema/snapshot";
import type { StateConfig } from "./config";

export type SessionState = {
  sessionId: string;
  paneKey: string;
  state: SessionPhase;
  activity: Activity | null;
  /** When `state` began. */
  since: number;
  lastEventAt: number;
  /** toolUseId of the pending permission while waiting. */
  waitingOn: string | null;
  /** toolUseId of the most recent tool.start on the parent (not a subagent). */
  currentToolUseId: string | null;
  backgroundTasks: number;
  /** Live subagent ids. An array, not a Set, so state is JSON-serialisable and comparable. */
  subagents: string[];
  contextPct: number | null;
  errorFlag: boolean;
  /** The previous state, shown until `until` so quick flips do not flicker. */
  displayHold: { state: SessionPhase; until: number } | null;
};

export function newSession(sessionId: string, paneKey: string, now: number): SessionState {
  return {
    sessionId,
    paneKey,
    state: "idle",
    activity: null,
    since: now,
    lastEventAt: now,
    waitingOn: null,
    currentToolUseId: null,
    backgroundTasks: 0,
    subagents: [],
    contextPct: null,
    errorFlag: false,
    displayHold: null,
  };
}

function transition(s: SessionState, to: SessionPhase, now: number, cfg: StateConfig): SessionState {
  if (s.state === to) return s;
  return { ...s, state: to, since: now, displayHold: { state: s.state, until: now + cfg.minDisplayMs } };
}

const busy = (s: SessionState): boolean => s.backgroundTasks > 0 || s.subagents.length > 0;
const resting = (s: SessionState): boolean => s.state === "idle" || s.state === "asleep";

/** Pure: applies one event to one session. `now` is the event time. */
export function reduceSession(prev: SessionState, e: RedactedEvent, now: number, cfg: StateConfig): SessionState {
  if (prev.state === "done") return prev;
  let s: SessionState = { ...prev, lastEventAt: now };
  switch (e.kind) {
    case "session.start":
      return s;
    case "session.end":
      return transition({ ...s, activity: null, waitingOn: null }, "done", now, cfg);
    case "presence":
      return e.alive ? s : transition({ ...s, activity: null, waitingOn: null }, "done", now, cfg);
    case "prompt":
      return transition({ ...s, activity: "think", waitingOn: null }, "working", now, cfg);
    case "tool.start":
      s = e.agentId ? { ...s, activity: "spawn" } : { ...s, activity: e.category, currentToolUseId: e.toolUseId };
      return resting(s) ? transition(s, "working", now, cfg) : s;
    case "tool.end":
      if (s.state === "waiting" && s.waitingOn === e.toolUseId) s = transition({ ...s, waitingOn: null }, "working", now, cfg);
      return { ...s, activity: e.agentId || busy(s) ? "spawn" : "think" };
    case "permission.request":
      if (s.state === "working" || s.state === "idle") return transition({ ...s, waitingOn: e.toolUseId ?? s.currentToolUseId }, "waiting", now, cfg);
      return s;
    case "permission.denied":
      return s.state === "waiting" ? transition({ ...s, waitingOn: null, activity: "think" }, "working", now, cfg) : s;
    case "stop":
      s = { ...s, backgroundTasks: e.backgroundTasks };
      if (s.state !== "working") return s;
      if (busy(s)) return { ...s, activity: "spawn" };
      return transition({ ...s, activity: null, currentToolUseId: null }, "idle", now, cfg);
    case "subagent.start": {
      const subagents = s.subagents.includes(e.agentId) ? s.subagents : [...s.subagents, e.agentId];
      return { ...s, subagents, activity: s.state === "working" ? "spawn" : s.activity };
    }
    case "subagent.stop":
      return { ...s, subagents: s.subagents.filter((id) => id !== e.agentId) };
    case "status":
      return e.contextPct === undefined ? s : { ...s, contextPct: e.contextPct };
    default:
      return s;
  }
}

/** Pure: applies the passage of time. Called by the host on an interval. */
export function tickSession(s: SessionState, now: number, cfg: StateConfig): SessionState {
  if (s.state === "idle" && now - s.since >= cfg.asleepAfterMs) return transition(s, "asleep", now, cfg);
  return s;
}
```

```ts
import type { RedactedEvent } from "../schema/events";
import type { StateConfig } from "./config";
import { newSession, reduceSession, tickSession, type SessionState } from "./session";

export type PaneState = { paneKey: string; firstSeenAt: number; sessionIds: string[] };

/** Plain records rather than Maps so a fleet is JSON-serialisable and trivially comparable in tests. */
export type Fleet = { sessions: Record<string, SessionState>; panes: Record<string, PaneState> };

export function emptyFleet(): Fleet {
  return { sessions: {}, panes: {} };
}

function withSession(f: Fleet, s: SessionState): Fleet {
  const pane = f.panes[s.paneKey] ?? { paneKey: s.paneKey, firstSeenAt: s.since, sessionIds: [] };
  const panes = pane.sessionIds.includes(s.sessionId)
    ? f.panes
    : { ...f.panes, [s.paneKey]: { ...pane, sessionIds: [...pane.sessionIds, s.sessionId] } };
  return { sessions: { ...f.sessions, [s.sessionId]: s }, panes };
}

/** Pure: routes an event to its session, creating the session and pane on first sight. */
export function reduceFleet(f: Fleet, e: RedactedEvent, now: number, cfg: StateConfig): Fleet {
  const existing = f.sessions[e.sessionId];
  if (!existing && e.kind === "presence" && !e.alive) return f;
  const s = existing ?? newSession(e.sessionId, e.paneKey, now);
  return withSession(f, reduceSession(s, e, now, cfg));
}

/** Pure: applies timers to every session and drops what has lingered long enough. */
export function tickFleet(f: Fleet, now: number, cfg: StateConfig): Fleet {
  const sessions: Record<string, SessionState> = {};
  for (const s of Object.values(f.sessions)) {
    const ticked = tickSession(s, now, cfg);
    if (ticked.state === "done" && now - ticked.since >= cfg.doneLingerMs) continue;
    sessions[ticked.sessionId] = ticked;
  }
  const panes: Record<string, PaneState> = {};
  for (const p of Object.values(f.panes)) {
    const sessionIds = p.sessionIds.filter((id) => id in sessions);
    if (sessionIds.length > 0) panes[p.paneKey] = { ...p, sessionIds };
  }
  return { sessions, panes };
}
```

```ts
import { ensurePaneAlias, ensureSessionAlias, type AliasMap } from "../alias/names";
import { SnapshotSchema, type SessionPhase, type SessionSnapshot, type Snapshot } from "../schema/snapshot";
import type { Fleet } from "./fleet";
import type { SessionState } from "./session";

/** The state to show right now: the held previous state until the hold expires. */
export function displayedState(s: SessionState, now: number): SessionPhase {
  return s.displayHold && now < s.displayHold.until ? s.displayHold.state : s.state;
}

/**
 * The only place a wire Snapshot is made, and the only place local identifiers are left behind.
 * Returns the alias map too, since new sessions or panes may have been given aliases.
 */
export function deriveSnapshot(f: Fleet, aliases: AliasMap, now: number): { snapshot: Snapshot; aliases: AliasMap } {
  let map = aliases;
  const panes = Object.values(f.panes)
    .sort((a, b) => a.firstSeenAt - b.firstSeenAt)
    .map((p) => {
      const r = ensurePaneAlias(map, p.paneKey);
      map = r.map;
      return { pane: r.alias, label: r.label, collision: false };
    });
  const sessions: SessionSnapshot[] = Object.values(f.sessions)
    .sort((a, b) => a.since - b.since)
    .map((s) => {
      const sr = ensureSessionAlias(map, s.sessionId);
      map = sr.map;
      const pr = ensurePaneAlias(map, s.paneKey);
      map = pr.map;
      const state = displayedState(s, now);
      return {
        alias: sr.alias,
        label: sr.label,
        pane: pr.alias,
        state,
        activity: state === "working" ? s.activity : null,
        subagents: s.subagents.length,
        contextPct: s.contextPct,
        since: new Date(s.since).toISOString(),
        updatedAt: new Date(s.lastEventAt).toISOString(),
      };
    });
  const snapshot = SnapshotSchema.parse({ version: 1, generatedAt: new Date(now).toISOString(), panes, sessions });
  return { snapshot, aliases: map };
}
```

```ts
import type { RedactedEvent } from "../schema/events";
import { TickerEventSchema, type TickerEvent } from "../schema/ticker";
import type { SessionState } from "./session";

/**
 * Zero, one or two ticker lines for an event: the event's own line, then a `state` line when the
 * reducer changed the session's state. This is the only source of ticker lines, so the ticker can
 * never carry more than the redacted event does.
 */
export function toTickerEvents(e: RedactedEvent, before: SessionState | null, after: SessionState): TickerEvent[] {
  const t = new Date(e.t).toISOString();
  const lines: TickerEvent[] = [];
  const own = ownLine(e, t);
  if (own) lines.push(own);
  if (before && before.state !== after.state) lines.push({ t, kind: "state", state: after.state });
  return lines.map((l) => TickerEventSchema.parse(l));
}

function ownLine(e: RedactedEvent, t: string): TickerEvent | null {
  switch (e.kind) {
    case "session.start":
      return { t, kind: "session.start" };
    case "session.end":
      return { t, kind: "session.end", reason: e.reason };
    case "prompt":
      return { t, kind: "prompt" };
    case "tool.start":
      return { t, kind: "tool.start", tool: e.tool, activity: e.category };
    case "tool.end":
      return { t, kind: "tool.end", tool: e.tool, activity: e.category, durationMs: e.durationMs };
    case "permission.request":
      return { t, kind: "permission", tool: e.tool };
    case "permission.denied":
      return e.tool ? { t, kind: "permission", tool: e.tool } : { t, kind: "permission" };
    case "stop":
      return { t, kind: "stop" };
    case "subagent.start":
      return { t, kind: "subagent.start", agentType: e.agentType };
    case "subagent.stop":
      return { t, kind: "subagent.stop", agentType: e.agentType };
    case "status":
      return e.contextPct === undefined ? null : { t, kind: "context", contextPct: e.contextPct };
    case "presence":
      return null;
    default:
      return null;
  }
}
```

```ts
import { z } from "zod";
import { RedactedEventSchema, type RedactedEvent } from "../schema/events";

export type Fixture = { name: string; events: RedactedEvent[] };

const HeaderSchema = z.object({ fixture: z.literal(1), name: z.string().min(1) }).strict();

/** One JSON header line, then one validated RedactedEvent per line. Blank lines are ignored. */
export function parseFixture(text: string): Fixture {
  const lines = text.split(/\r?\n/).filter((l) => l.trim().length > 0);
  const first = lines[0];
  if (!first) throw new Error("fixture is empty");
  const header = HeaderSchema.safeParse(JSON.parse(first));
  if (!header.success) throw new Error('fixture must start with a header line {"fixture":1,"name":...}');
  const events = lines.slice(1).map((line, i) => {
    const parsed = RedactedEventSchema.safeParse(JSON.parse(line));
    if (!parsed.success) throw new Error(`fixture ${header.data.name} line ${i + 2}: ${parsed.error.message}`);
    return parsed.data;
  });
  return { name: header.data.name, events };
}
```

```ts
import type { Fixture } from "../fixtures/load";
import type { Sink, Source } from "./source";

export type ReplayOptions = {
  /** 1 = recorded pace, 2 = twice as fast, Infinity = all at once. Default 1. */
  speed?: number;
  setTimeout?: (fn: () => void, ms: number) => unknown;
  clearTimeout?: (handle: unknown) => void;
};

/** Plays a fixture back with its recorded gaps scaled by `speed`. */
export function createReplaySource(fixture: Fixture, opts: ReplayOptions = {}): Source {
  const speed = opts.speed ?? 1;
  const setT = opts.setTimeout ?? ((fn, ms) => globalThis.setTimeout(fn, ms));
  const clearT = opts.clearTimeout ?? ((h) => globalThis.clearTimeout(h as ReturnType<typeof globalThis.setTimeout>));
  return {
    start(sink: Sink) {
      const events = fixture.events;
      if (!Number.isFinite(speed)) {
        for (const e of events) sink(e);
        return () => {};
      }
      let i = 0;
      let handle: unknown = null;
      let stopped = false;
      const emitNext = (): void => {
        if (stopped) return;
        const e = events[i];
        if (!e) return;
        sink(e);
        i++;
        const next = events[i];
        if (!next) return;
        handle = setT(emitNext, Math.max(0, (next.t - e.t) / speed));
      };
      emitNext();
      return () => {
        stopped = true;
        if (handle !== null) clearT(handle);
      };
    },
  };
}
```

```ts
import type { RedactedEvent, ToolCategory } from "../schema/events";
import type { StateConfig } from "../state/config";
import type { Sink, Source } from "./source";

/** Short timers so a demo shows sleeping and departures within a minute or two. */
export const SIMULATED_CONFIG: StateConfig = { asleepAfterMs: 20_000, minDisplayMs: 1_500, doneLingerMs: 4_000 };

/** mulberry32: small, seedable, good enough for a demo. */
function rng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Omit that distributes over a union, so each event variant keeps its own fields. */
type DistributiveOmit<T, K extends PropertyKey> = T extends unknown ? Omit<T, K> : never;
type EventBody = DistributiveOmit<RedactedEvent, "v" | "t" | "sessionId" | "paneKey">;

type Phase = "unborn" | "working" | "tooling" | "waiting" | "subagent" | "idle" | "gone";
type Sim = {
  id: string;
  paneKey: string;
  phase: Phase;
  nextAt: number;
  toolUseId: string;
  toolName: string;
  toolStartedAt: number;
  toolCategory: ToolCategory;
  agentId: string;
  idleSince: number;
  serial: number;
};

const TOOLS: [string, ToolCategory][] = [["Read", "read"], ["Grep", "read"], ["Edit", "edit"], ["Write", "edit"], ["Bash", "run"], ["WebFetch", "web"]];
const PANES = ["5a1e0000000000a1", "5a1e0000000000b2", "5a1e0000000000c3"];

export type SimulatedFleetOptions = { seed?: number; sessions?: number };

/**
 * A deterministic fleet that emits redacted events, so simulated mode runs the same reducers as
 * live mode. Call `next(now)` with a non-decreasing clock; it returns the events due by then.
 */
export function createSimulatedFleet(opts: SimulatedFleetOptions = {}): { next(now: number): RedactedEvent[] } {
  const random = rng(opts.seed ?? 1);
  const count = opts.sessions ?? 6;
  const between = (lo: number, hi: number): number => lo + random() * (hi - lo);
  let serial = 0;
  const sims: Sim[] = [];
  const fresh = (slot: number, bornAt: number): Sim => ({
    id: `sim-${slot}-${++serial}`,
    paneKey: PANES[slot % PANES.length]!,
    phase: "unborn",
    nextAt: bornAt,
    toolUseId: "",
    toolName: "",
    toolStartedAt: 0,
    toolCategory: "read",
    agentId: "",
    idleSince: 0,
    serial: 0,
  });
  let started = false;

  function emit(s: Sim, now: number, body: EventBody): RedactedEvent {
    return { v: 1, t: now, sessionId: s.id, paneKey: s.paneKey, ...body } as RedactedEvent;
  }

  function step(s: Sim, now: number, out: RedactedEvent[]): void {
    const r = random();
    switch (s.phase) {
      case "unborn":
        out.push(emit(s, now, { kind: "session.start", source: "startup" }), emit(s, now, { kind: "prompt" }));
        s.phase = "working";
        s.nextAt = now + between(800, 2500);
        return;
      case "working": {
        if (r < 0.5) {
          const [tool, category] = TOOLS[Math.floor(random() * TOOLS.length)]!;
          s.toolUseId = `sim-tool-${++s.serial}`;
          s.toolName = tool;
          s.toolStartedAt = now;
          s.toolCategory = category;
          out.push(emit(s, now, { kind: "tool.start", tool, category, toolUseId: s.toolUseId }));
          s.phase = "tooling";
          s.nextAt = now + between(300, 2500);
        } else if (r < 0.62) {
          s.toolUseId = `sim-tool-${++s.serial}`;
          s.toolName = "Write";
          s.toolCategory = "edit";
          out.push(
            emit(s, now, { kind: "tool.start", tool: "Write", category: "edit", toolUseId: s.toolUseId }),
            emit(s, now, { kind: "permission.request", tool: "Write", category: "edit" }),
          );
          s.phase = "waiting";
          s.nextAt = now + between(4000, 9000);
        } else if (r < 0.72) {
          s.agentId = `sim-agent-${++s.serial}`;
          out.push(emit(s, now, { kind: "subagent.start", agentId: s.agentId, agentType: "general-purpose" }));
          s.phase = "subagent";
          s.nextAt = now + between(4000, 8000);
        } else {
          out.push(emit(s, now, { kind: "stop", backgroundTasks: 0 }));
          s.phase = "idle";
          s.idleSince = now;
          s.nextAt = now + between(3000, 8000);
        }
        return;
      }
      case "tooling":
        out.push(emit(s, now, { kind: "tool.end", tool: s.toolName, category: s.toolCategory, toolUseId: s.toolUseId, durationMs: now - s.toolStartedAt }));
        s.phase = "working";
        s.nextAt = now + between(300, 1500);
        return;
      case "waiting":
        out.push(emit(s, now, { kind: "tool.end", tool: "Write", category: "edit", toolUseId: s.toolUseId, durationMs: 40 }));
        s.phase = "working";
        s.nextAt = now + between(500, 1500);
        return;
      case "subagent":
        out.push(emit(s, now, { kind: "subagent.stop", agentId: s.agentId, agentType: "general-purpose" }), emit(s, now, { kind: "prompt" }));
        s.agentId = "";
        s.phase = "working";
        s.nextAt = now + between(500, 1500);
        return;
      case "idle":
        if (r < 0.1) {
          out.push(emit(s, now, { kind: "session.end", reason: "prompt_input_exit" }));
          s.phase = "gone";
          s.nextAt = now + between(10000, 20000);
        } else if (r < 0.4 || now - s.idleSince > SIMULATED_CONFIG.asleepAfterMs + between(5000, 15000)) {
          out.push(emit(s, now, { kind: "prompt" }));
          s.phase = "working";
          s.nextAt = now + between(800, 2500);
        } else {
          s.nextAt = now + between(3000, 8000);
        }
        return;
      case "gone": {
        const slot = sims.indexOf(s);
        sims[slot] = fresh(slot, now);
        return;
      }
      default:
        return;
    }
  }

  return {
    next(now: number): RedactedEvent[] {
      if (!started) {
        started = true;
        for (let i = 0; i < count; i++) sims.push(fresh(i, now + i * between(2500, 5000)));
      }
      const out: RedactedEvent[] = [];
      for (let i = 0; i < sims.length; i++) {
        let guard = 0;
        while (sims[i]!.nextAt <= now && guard++ < 10) step(sims[i]!, now, out);
      }
      return out;
    },
  };
}

export type SimulatedSourceOptions = SimulatedFleetOptions & {
  /** Multiplier on the fleet's pace. Default 1. */
  speed?: number;
  now?: () => number;
  tickMs?: number;
  setInterval?: (fn: () => void, ms: number) => unknown;
  clearInterval?: (handle: unknown) => void;
};

/** Wraps the fleet in a timer so it behaves like any other Source. */
export function createSimulatedSource(opts: SimulatedSourceOptions = {}): Source {
  const now = opts.now ?? (() => Date.now());
  const tickMs = opts.tickMs ?? 250;
  const speed = opts.speed ?? 1;
  const setI = opts.setInterval ?? ((fn, ms) => globalThis.setInterval(fn, ms));
  const clearI = opts.clearInterval ?? ((h) => globalThis.clearInterval(h as ReturnType<typeof globalThis.setInterval>));
  return {
    start(sink: Sink) {
      const fleet = createSimulatedFleet(opts);
      const origin = now();
      const clock = (): number => origin + (now() - origin) * speed;
      for (const e of fleet.next(clock())) sink(e);
      const handle = setI(() => {
        for (const e of fleet.next(clock())) sink(e);
      }, tickMs);
      return () => clearI(handle);
    },
  };
}
```

```ts
import { emptyAliasMap } from "../alias/names";
import type { Snapshot } from "../schema/snapshot";
import { DEFAULT_CONFIG } from "../state/config";
import { emptyFleet, reduceFleet, tickFleet } from "../state/fleet";
import { deriveSnapshot } from "../state/snapshot";
import type { Fixture } from "./load";

export type SnapshotFrame = Omit<Snapshot, "generatedAt">;

/**
 * Reduce a fixture and record the snapshot after every event that changes something visible.
 * Each snapshot is taken after the display hold has expired so frames show real states, and
 * `updatedAt` changes on every event, so a frame is kept only when something else changed too.
 */
export function snapshotFrames(fixture: Fixture): SnapshotFrame[] {
  let fleet = emptyFleet();
  let aliases = emptyAliasMap();
  const frames: SnapshotFrame[] = [];
  let last = "";
  for (const e of fixture.events) {
    fleet = tickFleet(reduceFleet(fleet, e, e.t, DEFAULT_CONFIG), e.t, DEFAULT_CONFIG);
    const r = deriveSnapshot(fleet, aliases, e.t + DEFAULT_CONFIG.minDisplayMs);
    aliases = r.aliases;
    const { generatedAt: _generatedAt, ...frame } = r.snapshot;
    const key = JSON.stringify({ ...frame, sessions: frame.sessions.map(({ updatedAt: _u, ...s }) => s) });
    if (key !== last) {
      frames.push(frame);
      last = key;
    }
  }
  return frames;
}
```

```ts
// regenerates the golden snapshot sequences for every fixture.
//
//   pnpm --filter @clawd/core fixtures

// Raw recordings never enter the repo; they are gitignored under tools/hook-spike/.

import { join } from "node:path";
import { redact } from "../src/ingest/redact";
import { snapshotFrames } from "../src/fixtures/frames";
import { parseFixture } from "../src/fixtures/load";

const root = join(import.meta.dirname, "..", "..", "..");
const fixturesDir = join(root, "packages", "core", "src", "fixtures");
const recordings: [string, string][] = [
  ["spike-headless", join(root, "tools", "hook-spike", "spike-events-headless.jsonl")],
  ["spike-interactive", join(root, "tools", "hook-spike", "spike-events-interactive.jsonl")],
];
const BASE = 1_757_671_200_000;

type Recorded = { t: number; url: string; body: { kind?: string; payload?: unknown } & Record<string, unknown> };

/** The listener wrapped each POST as { t, url, body }; command hooks nested the real payload under body.payload. */
function payloadOf(rec: Recorded): unknown {
  return rec.body.kind === "command-hook" ? rec.body.payload : rec.body;
}

for (const [name, file] of recordings) {
  if (!existsSync(file)) {
    console.log(`skip ${name}: ${file} not found`);
    continue;
  }
  const lines = readFileSync(file, "utf8").split(/\r?\n/).filter((l) => l.trim().length > 0);
  const out: string[] = [JSON.stringify({ fixture: 1, name })];
  // The listener's clock restarts when it is relaunched; keep fixture time monotonic across restarts.
  let offset = 0;
  let lastT = -Infinity;
  let kept = 0;
  for (const line of lines) {
    const rec = JSON.parse(line) as Recorded;
    if (rec.t < lastT) offset += lastT + 1000;
    lastT = rec.t;
    const payload = payloadOf(rec);
    const cwd = typeof payload === "object" && payload !== null ? (payload as { cwd?: unknown }).cwd : undefined;
    const e = redact(payload, { now: BASE + offset + rec.t, allowlist: typeof cwd === "string" ? [cwd] : [] });
    if (e) {
      out.push(JSON.stringify(e));
      kept++;
    }
  }
  writeFileSync(join(fixturesDir, `${name}.jsonl`), out.join("\n") + "\n");
  console.log(`${name}: ${kept} events from ${lines.length} recorded lines`);
}

mkdirSync(join(fixturesDir, "golden"), { recursive: true });
for (const name of ["permission-denied", "killed-session", "spike-headless", "spike-interactive"]) {

  writeFileSync(join(fixturesDir, "golden", `${name}.json`), JSON.stringify(frames, null, 1) + "\n");
  console.log(`golden ${name}: ${frames.length} frames`);
}
```
